**Class declaration:**
```swift
import Foundation
import FirebaseFirestore
import Combine

@MainActor
final class ActivityViewModel: ObservableObject {
    @Published var activeActivities: [Activity] = []
    @Published var archivedActivities: [Activity] = []
    @Published var isLoading = false
    @Published var errorMessage: String?

    private let repository: ActivityRepositoryProtocol
    private let userId: String

    // CRITICAL: Store listeners to remove in deinit (prevents memory leaks)
    private var activeListener: ListenerRegistration?
    private var archivedListener: ListenerRegistration?

    init(userId: String, repository: ActivityRepositoryProtocol = ActivityRepository()) {
        self.userId = userId
        self.repository = repository
    }
}
```

**Methods to implement:**

1. `func startListening()` - Attach Firestore listeners
   - Call repository.listenToActiveActivities with [weak self] closure
   - Call repository.listenToArchivedActivities with [weak self] closure
   - Store returned ListenerRegistration in activeListener and archivedListener properties
   - Update @Published arrays in closures (automatically triggers UI refresh)

2. `func createActivity(name: String, category: ActivityCategory) async`
   - Set isLoading = true at start, defer { isLoading = false }
   - Create Activity with nil id, trimmed name, category.rawValue, ISO 8601 timestamps, archived=false
   - Call repository.createActivity
   - Catch errors and set errorMessage (user-friendly description)
   - Listener automatically updates activeActivities (no manual array update needed)

3. `func updateActivity(_ activity: Activity, name: String, category: ActivityCategory) async`
   - Create updated Activity with new name/category, updated timestamp
   - Call repository.updateActivity
   - Catch errors and set errorMessage
   - Listener automatically updates activeActivities

4. `func archiveActivity(_ activity: Activity) async`
   - Guard let activityId = activity.id
   - Call repository.archiveActivity(userId, activityId)
   - Catch errors and set errorMessage
   - Listener automatically moves activity from active to archived list

5. `func restoreActivity(_ activity: Activity) async`
   - Guard let activityId = activity.id
   - Call repository.restoreActivity(userId, activityId)
   - Catch errors and set errorMessage
   - Listener automatically moves activity from archived to active list

6. `func deleteActivity(_ activity: Activity) async`
   - Guard let activityId = activity.id
   - Call repository.deleteActivity(userId, activityId)
   - Catch errors and set errorMessage
   - Listener automatically removes activity from list

7. `deinit { /* cleanup */ }`
   - Call activeListener?.remove()
   - Call archivedListener?.remove()
   - Add print("ActivityViewModel deinitialized") for debugging memory management

CRITICAL patterns from research:
- ALWAYS use [weak self] in listener closures to prevent retain cycles
- ALWAYS remove listeners in deinit (memory leak prevention from STATE.md pitfall #4)
- Do NOT manually update @Published arrays after operations - listeners handle it automatically
- Use defer { isLoading = false } to ensure loading state clears even on error

**2. ActivityListView.swift - Active activities with create/edit/archive/delete:**
```swift
import SwiftUI

struct ActivityListView: View {
    @StateObject private var viewModel: ActivityViewModel
    @State private var showingCreateSheet = false
    @State private var editingActivity: Activity?

    init(userId: String) {
        _viewModel = StateObject(wrappedValue: ActivityViewModel(userId: userId))
    }

    var body: some View {
        NavigationStack {
            List {
                ForEach(viewModel.activeActivities) { activity in
                    ActivityRowView(activity: activity)
                        .onTapGesture {
                            editingActivity = activity
                        }
                        .swipeActions(edge: .trailing, allowsFullSwipe: false) {
                            // Delete (leftmost, red)
                            Button(role: .destructive) {
                                Task { await viewModel.deleteActivity(activity) }
                            } label: {
                                Label("Delete", systemImage: "trash")
                            }

                            // Archive (rightmost, orange)
                            Button {
                                Task { await viewModel.archiveActivity(activity) }
                            } label: {
                                Label("Archive", systemImage: "archivebox")
                            }
                            .tint(.orange)
                        }
                }
            }
            .navigationTitle("Activities")
            .toolbar {
                ToolbarItem(placement: .primaryAction) {
                    Button {
                        showingCreateSheet = true
                    } label: {
                        Image(systemName: "plus")
                    }
                }

                ToolbarItem(placement: .navigationBarLeading) {
                    NavigationLink {
                        ArchivedActivityListView(viewModel: viewModel)
                    } label: {
                        Label("Archived", systemImage: "archivebox")
                    }
                }
            }
            .sheet(isPresented: $showingCreateSheet) {
                ActivityFormView(activity: nil, onSave: { name, category in
                    Task {
                        await viewModel.createActivity(name: name, category: category)
                    }
                })
            }
            .sheet(item: $editingActivity) { activity in
                ActivityFormView(activity: activity, onSave: { name, category in
                    Task {
                        await viewModel.updateActivity(activity, name: name, category: category)
                    }
                })
            }
            .overlay {
                if viewModel.activeActivities.isEmpty && !viewModel.isLoading {
                    ContentUnavailableView(
                        "No Activities",
                        systemImage: "music.note.list",
                        description: Text("Tap + to create your first practice activity")
                    )
                }
            }
        }
        .onAppear {
            viewModel.startListening()
        }
    }
}
```

- Call startListening() in onAppear, NOT in init (listeners attach when view appears)
