   ```swift
   import GoogleSignIn
   import FirebaseCore

   func signInWithGoogle() async throws -> User {
       guard let clientID = FirebaseApp.app()?.options.clientID else {
           throw AuthError.unknown("Firebase client ID not found")
       }

       let config = GIDConfiguration(clientID: clientID)
       GIDSignIn.sharedInstance.configuration = config

       // Get root view controller for presenting Google Sign-In
       guard let windowScene = await UIApplication.shared.connectedScenes.first as? UIWindowScene,
             let rootViewController = await windowScene.windows.first?.rootViewController else {
           throw AuthError.unknown("No root view controller found")
       }

       return try await withCheckedThrowingContinuation { continuation in
           GIDSignIn.sharedInstance.signIn(withPresenting: rootViewController) { result, error in
               if let error = error {
                   continuation.resume(throwing: AuthError.unknown(error.localizedDescription))
                   return
               }

               guard let user = result?.user,
                     let idToken = user.idToken?.tokenString else {
                   continuation.resume(throwing: AuthError.unknown("Google Sign-In failed"))
                   return
               }

               let credential = GoogleAuthProvider.credential(
                   withIDToken: idToken,
                   accessToken: user.accessToken.tokenString
               )

               Task {
                   do {
                       let result = try await Auth.auth().signIn(with: credential)
                       let user = User(from: result.user)

                       // Save user profile to Firestore (auto-creates if new)
                       let userRepo = UserRepository()
                       try await userRepo.saveUser(user)

                       continuation.resume(returning: user)
                   } catch {
                       continuation.resume(throwing: error)
                   }
               }
           }
       }
   }
   ```

   ```swift
   import AuthenticationServices

   func signInWithApple(idToken: String, nonce: String, fullName: PersonNameComponents?) async throws -> User {
       // CRITICAL: Use OAuthProvider.appleCredential with fullName parameter
       // This preserves display name on first sign-in (Apple only provides it once)

       do {
           let result = try await Auth.auth().signIn(with: credential)
           let user = User(from: result.user)

           // Save user profile to Firestore
           let userRepo = UserRepository()
           try await userRepo.saveUser(user)

           return user
       } catch let error as NSError {
           throw mapFirebaseError(error)
       }
   }
   ```

3. Update AuthViewModel.swift to add OAuth methods:

   Add imports:
   ```swift
   import AuthenticationServices
   ```

   Add nonce property and helpers:
   ```swift
   private var currentNonce: String?

   private func randomNonceString(length: Int = 32) -> String {
       let charset = Array("0123456789ABCDEFGHIJKLMNOPQRSTUVXYZabcdefghijklmnopqrstuvwxyz-._")
       return String((0..<length).map { _ in charset.randomElement()! })
   }

   private func sha256(_ input: String) -> String {
       let inputData = Data(input.utf8)
       let hashedData = SHA256.hash(data: inputData)
       return hashedData.compactMap { String(format: "%02x", $0) }.joined()
   }
   ```

   Add Google Sign-In method:
   ```swift
   func signInWithGoogle() async {
       isLoading = true
       errorMessage = nil
       defer { isLoading = false }

       do {
           user = try await repository.signInWithGoogle()
       } catch {
           errorMessage = error.localizedDescription
       }
   }
   ```

   Add Sign in with Apple method and delegate conformance:
   ```swift
   // Add NSObject and ASAuthorizationControllerDelegate conformance to class declaration
   // Change: final class AuthViewModel: ObservableObject
   // To: final class AuthViewModel: NSObject, ObservableObject, ASAuthorizationControllerDelegate

   func signInWithApple() {
       let nonce = randomNonceString()
       currentNonce = nonce

       let provider = ASAuthorizationAppleIDProvider()
       let request = provider.createRequest()
       request.requestedScopes = [.fullName, .email]
       request.nonce = sha256(nonce)

       let controller = ASAuthorizationController(authorizationRequests: [request])
       controller.delegate = self
       controller.performRequests()
   }

   // ASAuthorizationControllerDelegate methods
   func authorizationController(controller: ASAuthorizationController,
                               didCompleteWithAuthorization authorization: ASAuthorization) {
       guard let appleIDCredential = authorization.credential as? ASAuthorizationAppleIDCredential,
             let nonce = currentNonce,
             let appleIDToken = appleIDCredential.identityToken,
             let idTokenString = String(data: appleIDToken, encoding: .utf8) else {
           errorMessage = "Apple Sign-In failed"
           return
       }

       Task {
           isLoading = true
           errorMessage = nil
           defer { isLoading = false }

           do {
               user = try await repository.signInWithApple(
                   idToken: idTokenString,
                   nonce: nonce,
                   fullName: appleIDCredential.fullName
               )
           } catch {
               errorMessage = error.localizedDescription
           }
       }
   }

   func authorizationController(controller: ASAuthorizationController,
                               didCompleteWithError error: Error) {
       errorMessage = error.localizedDescription
   }
   ```

CRITICAL: For Apple Sign-In, generate cryptographically secure nonce with randomNonceString(), send SHA256 hash to Apple (not raw nonce), use raw nonce with Firebase. Use OAuthProvider.appleCredential(withIDToken:rawNonce:fullName:) NOT OAuthProvider.credential() - the fullName parameter preserves display name on first sign-in (Apple only provides it once). For Google Sign-In, use withCheckedThrowingContinuation to bridge callback-based GIDSignIn API to async/await. Per CONTEXT.md: auto-create accounts on first OAuth sign-in, accept Apple private relay emails.
