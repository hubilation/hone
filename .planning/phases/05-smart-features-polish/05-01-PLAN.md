    ```swift
    import Foundation

    /// Stateless utility for computing activity suggestions and practice streak.
    /// Pattern: mirrors ActivityGrouping.swift — no state, no dependencies, pure functions.
    struct SuggestionsService {

        // MARK: - Suggestion Scoring (D-01)

        /// Returns top N activities ranked by how overdue they are.
        /// Returns empty array when no activity has a lastUsed date (new user — no history to suggest from).
        static func suggestedActivities(
            activities: [Activity],
            sessions: [Session],
            limit: Int = 4
        ) -> [Activity] {
            let now = Date()
            let calendar = Calendar.current

            let candidates = activities.filter { $0.isActive && !$0.isCompleted }

            // New-user guard: if no activity has been practiced, show nothing
            let anyPracticed = candidates.contains { $0.lastUsed != nil }
            guard anyPracticed else { return [] }

            func score(_ activity: Activity) -> Double {
                // Recency: days since last practiced. Never-practiced = very high score.
                let daysSinceLast: Double
                if let lastUsedStr = activity.lastUsed,
                   let date = Date(iso8601String: lastUsedStr) {
                    let days = calendar.dateComponents([.day], from: date, to: now).day ?? 0
                    daysSinceLast = Double(max(0, days))
                } else {
                    daysSinceLast = 365.0
                }

                // Frequency proxy: invert recency over 30-day window.
                // Activities practiced a long time ago get low frequency score (haven't been frequent).
                let frequencyScore = max(0.0, 30.0 - daysSinceLast)

                let recencyWeight = 0.7
                let frequencyWeight = 0.3
                return recencyWeight * daysSinceLast + frequencyWeight * frequencyScore
            }

            return candidates
                .sorted { score($0) > score($1) }
                .prefix(limit)
                .map { $0 }
        }

        // MARK: - Streak Computation (D-03)

        /// Returns consecutive calendar days with at least one ended session.
        /// If today has no session yet, preserves yesterday's streak (today may not be over).
        /// Returns 0 when no sessions exist.
        static func currentStreak(sessions: [Session]) -> Int {
            let calendar = Calendar.current
            let today = calendar.startOfDay(for: Date())

            let practicedDays: Set<Date> = Set(
                sessions
                    .filter { $0.state == "ended" }
                    .compactMap { session -> Date? in
                        guard let date = Date(iso8601String: session.startTime) else { return nil }
                        return calendar.startOfDay(for: date)
                    }
            )

            guard !practicedDays.isEmpty else { return 0 }

            // If today has no session, start from yesterday (streak is alive until day ends)
            var checkDay = practicedDays.contains(today)
                ? today
                : calendar.date(byAdding: .day, value: -1, to: today)!

            var streak = 0
            while practicedDays.contains(checkDay) {
                streak += 1
                checkDay = calendar.date(byAdding: .day, value: -1, to: checkDay)!
            }

            return streak
        }
    }
    ```
