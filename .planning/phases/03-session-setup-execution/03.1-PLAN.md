1. Query all sessions collection for the user
2. For each session, query activities subcollection where activityId == targetActivityId
3. Filter where notes != nil and !notes.isEmpty
4. Sort by createdAt descending
5. Limit to requested number

```swift
func getHistoricalNotes(userId: String, activityId: String, limit: Int = 10) async throws -> [SessionActivity] {
    // Query strategy: Use collection group query to search across all session activities
    let snapshot = try await db.collectionGroup("activities")
        .whereField("activityId", isEqualTo: activityId)
        .order(by: "createdAt", descending: true)
        .limit(to: limit)
        .getDocuments()
    
    let activities = snapshot.documents.compactMap { doc -> SessionActivity? in
        do {
            let activity = try doc.data(as: SessionActivity.self)
            // Filter: only return activities with notes and not in-between time
            guard let notes = activity.notes, 
                  !notes.isEmpty, 
                  !activity.isInBetweenTime else {
                return nil
            }
            return activity
        } catch {
            print("Error decoding SessionActivity: \(error)")
            return nil
        }
    }
    
    return activities
}
```

**Key decisions:**
- Use collectionGroup query to search across all sessions efficiently
- Filter out activities without notes and in-between time activities
- Sort by createdAt descending (most recent first)
- Default limit of 10 notes (UI will show 3 by default with expand)
- Use compactMap to skip activities without notes or decode errors

```swift
private func formatDate(_ isoString: String) -> String {
    guard let date = isoString.toDate() else {
        return isoString

    formatter.unitsStyle = .abbreviated
    return formatter.localizedString(for: date, relativeTo: Date())
```
